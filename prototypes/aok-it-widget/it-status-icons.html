Config-Referenz

Auszug aus dem tchop Theme-JSON. statusIcons mappt Status-Tag-Namen auf Icons.

// elements.itWidget
{
  "enabled": true,
  "title": "IT Meldungen",
  "connectedMixId": 4217,
  "statusIcons": {
    "Neu":             "alert-circle-filled",   // filled circle + exclamation
    "In Bearbeitung":  "clock-circle-filled",   // filled circle + clock
    "In Planung":       "calendar-circle-filled",  // filled circle + calendar
    "Gelöst":           "check-circle-filled"    // filled circle + checkmark
  },
  "statusColors": {
    // All statuses use the org's primary color.
    // Shape alone distinguishes state.
    "Neu":             { "icon": "colors.primary", "bg": "#E8F4F2", "text": "colors.primary" },
    "In Bearbeitung":  { "icon": "colors.primary", "bg": "#E8F4F2", "text": "colors.primary" },
    "In Planung":      { "icon": "colors.primary", "bg": "#E8F4F2", "text": "colors.primary" },
    "Gelöst":           { "icon": "colors.primary", "bg": "#E8F4F2", "text": "colors.primary" }
  }
}
